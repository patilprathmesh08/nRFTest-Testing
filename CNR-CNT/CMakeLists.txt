cmake_minimum_required(VERSION 3.20.0)

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(sta)

target_include_directories(app PUBLIC ${ZEPHYR_BASE}/subsys/net/ip)

target_sources(app PRIVATE
	src/main.c
	src/button_inputs.c
	src/epoch.c
	src/wifi_net.c
	src/joystick.c
	src/led.c
	src/frame.c
	src/ws.c
	src/https_conn.c
)

# Send-trigger button, compiled in ONLY when a button send-trigger mode is
# selected - see the SEND_TRIGGER choice in Kconfig.
# Tests the derived CONFIG_SEND_TRIGGER_NEEDS_BUTTON rather than the two
# button modes individually, so adding a third mode later cannot leave
# this line out of sync. A continuous-stream build carries no button code
# at all: the trigger is a debug/bench aid, not core telemetry.
target_sources_ifdef(CONFIG_SEND_TRIGGER_NEEDS_BUTTON app PRIVATE
	src/button.c
)

# Recorded-trajectory replay, compiled in ONLY when CONFIG_DEBUG_TFLITE is
# enabled. Kept conditional on purpose: replay_data.c is a ~13.8 KB const
# table, and a normal build should not carry it. The file is generated by
# gen_replay_data.py in the project's test folder - run it to regenerate,
# or with --check to verify it is still in sync with the source CSV.
target_sources_ifdef(CONFIG_DEBUG_TFLITE app PRIVATE
	src/replay_data.c
)
